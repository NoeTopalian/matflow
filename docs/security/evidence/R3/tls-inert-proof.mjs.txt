// R3: is the one accepted literal sslmode=require inert under E3's explicit config? A local fake Postgres speaks the
// SSLRequest handshake and presents a SELF-SIGNED certificate for the TEST hostname; DNS for that name is pointed at
// 127.0.0.1 inside this process only. Nothing reaches Neon. The URL carries synthetic credentials.
import fs from "node:fs"; import net from "node:net"; import tls from "node:tls"; import dns from "node:dns"; import path from "node:path";
import { pathToFileURL } from "node:url"; import { createRequire } from "node:module";
const [,, wt, dir] = process.argv;
const pg = createRequire(path.join(wt, "package.json"))("pg");
const blob = await import(pathToFileURL(path.join(wt, "scripts/readiness/ops-blob-host-count.mjs")).href);
const HOST = "ep-hidden-salad-abom7cg4.eu-west-2.aws.neon.tech";
const PORT = 55433;
const key = fs.readFileSync(path.join(dir, "self.key")); const cert = fs.readFileSync(path.join(dir, "self.crt"));
let handshakes = 0;
const server = net.createServer((sock) => {
  sock.once("data", (buf) => {
    if (buf.length === 8 && buf.readInt32BE(4) === 80877103) {
      sock.write("S");
      const t = new tls.TLSSocket(sock, { isServer: true, key, cert });
      t.on("secure", () => { handshakes += 1; t.end(); });
      t.on("error", () => {});
    } else sock.end();
  });
  sock.on("error", () => {});
});
await new Promise((r) => server.listen(PORT, "127.0.0.1", r));
const origLookup = dns.lookup;
dns.lookup = (h, o, cb) => {
  if (typeof o === "function") { cb = o; o = {}; }
  if (h === HOST) return o && o.all ? cb(null, [{ address: "127.0.0.1", family: 4 }]) : cb(null, "127.0.0.1", 4);
  return origLookup(h, o, cb);
};
async function attempt(label, cfg) {
  const before = handshakes;
  const c = new pg.Client({ ...cfg, connectionTimeoutMillis: 5000 });
  let out;
  try { await c.connect(); out = "CONNECTED"; } catch (e) { out = `ERROR ${e.code ?? ""} ${String(e.message).slice(0, 70)}`; }
  try { await c.end(); } catch {}
  console.log(JSON.stringify({ case: label, result: out, tlsHandshakeCompletedOnServer: handshakes > before }));
}
const url = `postgresql://synthetic:synthetic-pass@${HOST}:${PORT}/neondb?sslmode=require&channel_binding=require`;
const cfg = blob.buildClientConfig(url, { expectHost: "ep-hidden-salad", env: {} });
console.log(JSON.stringify({ builtConfig: { host: cfg.host === HOST, port: cfg.port, ssl: { rejectUnauthorized: cfg.ssl.rejectUnauthorized, servername: cfg.ssl.servername === HOST }, connectionString: "connectionString" in cfg } }));
await attempt("E3 explicit config from a URL with sslmode=require (self-signed cert)", cfg);
process.env.NODE_TLS_REJECT_UNAUTHORIZED = "0";
await attempt("same, with NODE_TLS_REJECT_UNAUTHORIZED=0 in the environment", cfg);
delete process.env.NODE_TLS_REJECT_UNAUTHORIZED;
await attempt("control: connectionString with sslmode=no-verify (what R2 showed the old guard let through)", { connectionString: `postgresql://synthetic:synthetic-pass@${HOST}:${PORT}/neondb?sslmode=no-verify` });
await attempt("control: connectionString with sslmode=require only (pg 8 treats require as verify-full)", { connectionString: url });
server.close();
dns.lookup = origLookup;
