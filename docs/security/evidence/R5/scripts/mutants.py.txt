def run(tests, tag):
    p = subprocess.run("npx vitest run " + " ".join(tests), cwd=WT, env=env, shell=True, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=600)
    out = p.stdout + p.stderr
    open(os.path.join(LOGS, "mutant-" + tag + ".log"), "w", encoding="utf-8").write(out)
    m = re.search(r"Tests\s+(.*)", out)
    fails = re.findall(r"(?:FAIL|×)\s+(.*)", out)
    return p.returncode, (m.group(1).strip() if m else "?"), fails[:6]

results = []
code, summ, _ = run(CSP_TESTS + REPORT_TESTS + UPLOAD_TESTS + ["tests/unit/sec-c2-blob-image-inline.test.ts", "tests/unit/sec-c2-request-validation.test.ts", "tests/unit/sec-e3-upload-limit-input-pixels.test.ts"], "baseline")
results.append({"mutant": "baseline (no mutation)", "exit": code, "tests": summ})
print("baseline", code, summ, flush=True)
for i, (name, rel, old, new, tests) in enumerate(MUTANTS):
    path = os.path.join(WT, rel)
    src = open(path, encoding="utf-8").read()
    if src.count(old) != 1:
        results.append({"mutant": name, "error": "anchor not found exactly once: %d" % src.count(old)})
        print(name, "ANCHOR MISSING", flush=True)
        continue
    open(path, "w", encoding="utf-8", newline="").write(src.replace(old, new))
    try:
        code, summ, fails = run(tests, "M%02d" % i)
    finally:
        subprocess.run(["git", "checkout", "--", rel], cwd=WT)
    verdict = "KILLED" if code != 0 else "SURVIVED"
    results.append({"mutant": name, "file": rel, "exit": code, "tests": summ, "verdict": verdict, "failing": fails})
    print(verdict, name, "|", summ, flush=True)
dirty = subprocess.run(["git", "status", "--porcelain", "--", "proxy.ts", "lib", "app"], cwd=WT, capture_output=True, text=True).stdout
results.append({"tree_after": dirty or "clean"})
json.dump(results, open(os.path.join(LOGS, "mutants.json"), "w"), indent=1)
print("tree after:", dirty or "clean")
