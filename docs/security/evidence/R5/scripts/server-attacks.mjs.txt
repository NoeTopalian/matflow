function raw(head, body, timeoutMs = 15000) {
  return new Promise((resolve) => {
    const sock = net.connect(Number(PORT), "127.0.0.1");
    let data = "";
    const t = setTimeout(() => { sock.destroy(); resolve({ timeout: true, statusLines: [...data.matchAll(/HTTP\/1\.1 (\d{3})/g)].map((m) => m[1]) }); }, timeoutMs);
    sock.on("data", (d) => { data += d.toString("latin1"); });
    sock.on("close", () => { clearTimeout(t); resolve({ timeout: false, statusLines: [...data.matchAll(/HTTP\/1\.1 (\d{3})/g)].map((m) => m[1]) }); });
    sock.on("error", (e) => { clearTimeout(t); resolve({ error: e.code, statusLines: [...data.matchAll(/HTTP\/1\.1 (\d{3})/g)].map((m) => m[1]) }); });
    sock.write(head);
    if (body) sock.write(body);
  });
}
