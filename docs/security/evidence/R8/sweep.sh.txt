#!/usr/bin/env bash
B=<scratchpad>/security/R8
S=$B/logs/sweep/summary.txt
echo "sha=$(git -C $B/wt rev-parse HEAD) start=$(date -u +%FT%TZ)" > $S
for f in sec-m1-demo-rotate-db sec-c3-rate-limit-atomic-db sec-c4-rate-limit-saturation-db sec-a4-lockout-burst-db attendance-import-engine; do
  for role in owner restricted; do
    bash $B/run-$role.sh tests/integration/$f.test.ts > $B/logs/sweep/$f.$role.log 2>&1
    rc=$?
    t=$(grep -E "^\s+Tests " $B/logs/sweep/$f.$role.log | tail -1 | sed 's/^ *//')
    echo "$f $role exit=$rc $t $(date -u +%T)" >> $S
  done
done
echo "done $(date -u +%FT%TZ)" >> $S
