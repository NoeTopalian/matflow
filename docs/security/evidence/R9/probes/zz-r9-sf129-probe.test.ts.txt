function shortName(dir: string, long: string): string | null {
  try {
    const t = execSync(`cmd /c dir /x /a "${dir}"`, { encoding: "utf8" });
    for (const l of t.split(/\r?\n/)) {
      const m = /\s(\S+~\d\S*)\s+(\S.*)$/.exec(l);
      if (m && m[2].trim() === long) return m[1];
    }
  } catch { /* ignore */ }
  return null;
}
function guard(p: string, prod: string[]) {
  try {
    const r = refuseProductionEnvFile(p, { productionEnvPaths: prod });
    return `ADMITTED ${path.basename(r)}`;
  } catch (e) {
    return e instanceof Refused ? e.message : `ERROR ${(e as Error).message}`;
  }
}
function spawnTool(args: string[], env: Record<string, string> = {}) {
  const clean = Object.fromEntries(Object.entries(process.env).filter(([k]) => !/^PG[A-Z]/i.test(k)));
  const r = spawnSync(process.execPath, [SCRIPT, ...args], { env: { ...clean, ...env } as unknown as NodeJS.ProcessEnv, encoding: "utf8", timeout: 30_000 });
  return { status: r.status, stdout: r.stdout, stderr: r.stderr.trim() };
}

describe("R9 SF-129 (a) env-file guard, real filesystem", () => {
  it("name, case, trailing dot/space, 8.3 short name, dot segment, junction, hard link, symlink", () => {
    const prod = [path.join(LAB, ".env")];
    const r: Record<string, string> = {};
    for (const n of [".ENV", ".Env", ".eNV", ".env.", ".env ", ".env. .", ".env::$DATA"]) r[`name ${JSON.stringify(n)}`] = guard(path.join(LAB, n), prod);
    const sn = shortName(LAB, ".env");
    out.shortName = sn;
    if (sn) r[`8.3 short name ${sn}`] = guard(path.join(LAB, sn), prod);
    r["dot segment .env\\."] = guard(`${path.join(LAB, ".env")}\\.`, prod);
    r["dot segment .env/."] = guard(`${path.join(LAB, ".env")}/.`, prod);
    const J = path.join(LABROOT, `${path.basename(LAB)}-junc`);
    fs.symlinkSync(LAB, J, "junction");
    r["junction/.env"] = guard(path.join(J, ".env"), prod);
    r["junction/.ENV"] = guard(path.join(J, ".ENV"), prod);
    if (sn) r[`junction/${sn}`] = guard(path.join(J, sn), prod);
    fs.linkSync(path.join(LAB, ".env"), path.join(LAB, "hard.txt"));
    r["hard link hard.txt"] = guard(path.join(LAB, "hard.txt"), prod);
    r["hard link via junction"] = guard(path.join(J, "hard.txt"), prod);
    const hsn = shortName(LAB, "hard.txt");
    try {
      fs.symlinkSync(path.join(LAB, ".env"), path.join(LAB, "sym.txt"), "file");
      r["file symlink sym.txt"] = guard(path.join(LAB, "sym.txt"), prod);
    } catch (e) {
      r["file symlink sym.txt"] = `NOT CREATED: ${(e as NodeJS.ErrnoException).code}`;
    }
    r["positive control ok.env"] = guard(path.join(LAB, "ok.env"), prod);
    out.guard = { ...r, hardShort: hsn };
    for (const [k, v] of Object.entries(r)) {
      if (k.startsWith("positive")) expect(v, k).toBe("ADMITTED ok.env");
      else if (v.startsWith("NOT CREATED")) continue;
      else expect(v.startsWith("ADMITTED"), `${k}: ${v}`).toBe(false);

  });

  it("spawned tool (cwd = worktree, decoy .env at the worktree root): every alias refused BEFORE the file is read", () => {

describe("R9 SF-129 (b) --confirm without --ids", () => {
  it("exit 2, before reading the env file or connecting", () => {
    const t = spawnTool(["--slug", "secr9-x", "--env-file", path.join(LAB, "ok.env"), "--confirm"]);
    const t2 = spawnTool(["--slug", "secr9-x", "--confirm", "--env-file", path.join(LAB, "missing.env")]);
    out.confirmNoIds = { a: { status: t.status, stderr: t.stderr }, missingEnvFile: { status: t2.status, stderr: t2.stderr } };
    // t2 (missing env file): Node 24 itself scans argv for --env-file even after the script name and exits 9 (recorded, finding)
    for (const x of [t]) {
      expect(x.status).toBe(2);
      expect(x.stderr).toMatch(/--confirm requires --ids/);
      expect(x.stdout).not.toMatch(/target:/);
    }
    expect(() => parseArgs(["--slug", "secr9-x", "--confirm", "--ids", ""])).toThrow(Refused);
    expect(() => parseArgs(["--slug", "secr9-x", "--confirm", "--ids", ","])).toThrow(Refused);
  });
});
