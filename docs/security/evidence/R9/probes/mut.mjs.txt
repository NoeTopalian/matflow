function run(kind, files) {
  const runner = kind === U ? `${B}/run-unit-wtm.sh` : `${B}/run-owner-wtm.sh`;
  const r = spawnSync("bash", [runner, ...files], { encoding: "utf8", timeout: 900_000 });
  const out = `${r.stdout}\n${r.stderr}`;
  const tests = (out.match(/^\s+Tests\s+.*$/m) ?? [""])[0].trim();
  const fails = [...new Set([...out.matchAll(/FAIL\s+(.+?)(?:\s\[|$)/gm)].map((m) => m[1].trim()))].slice(0, 12);
  return { kind, files, exit: r.status, tests, fails, out };
}
