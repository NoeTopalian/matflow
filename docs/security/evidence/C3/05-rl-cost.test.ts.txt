async function oldPath(bucket: string, max: number, windowMs: number) {
  const cutoff = new Date(Date.now() - windowMs);
  const agg = await prisma.rateLimitHit.groupBy({ by: ["bucket"], where: { bucket, hitAt: { gte: cutoff } }, _count: { _all: true }, _min: { hitAt: true } });
  if ((agg[0]?._count?._all ?? 0) >= max) return false;
  await prisma.rateLimitHit.create({ data: { bucket } });
  return true;
}
const med = (a: number[]) => { const s = [...a].sort((x, y) => x - y); return Math.round(s[Math.floor(s.length / 2)] * 10) / 10; };
async function time(n: number, f: (i: number) => Promise<unknown>) {
  const t: number[] = [];
  for (let i = 0; i < n; i++) { const s = performance.now(); await f(i); t.push(performance.now() - s); }
  return { median: med(t), min: Math.round(Math.min(...t) * 10) / 10 };
}
