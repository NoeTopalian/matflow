// classification JSON, a per-file markdown table and totals. Classes:
//   a = runs as matflow_ops through a separate client/URL (operator console, cron, platform-level audit)

const out = [];
for (const s of sites) {
  const key = `${s.file}:${s.line}`;
  let cls, note;
  if (site[key]) [cls, note] = site[key];
  else {
    const fc = fileClass.find(([re]) => re.test(s.file));
    if (!fc) throw new Error(`unclassified ${key}`);
    cls = fc[1]; note = fc[2];
  }
  out.push({ ...s, cls, note });
}
const unused = Object.keys(site).filter((k) => !out.some((o) => `${o.file}:${o.line}` === k));
if (unused.length) throw new Error(`override keys that match no site: ${unused.join(", ")}`);
fs.writeFileSync(`${outDir}/callsite-classification.json`, JSON.stringify(out, null, 1));

const totals = {};
const filesPer = {};
for (const o of out) {
  const c = o.cls.replace("*", "");
  totals[c] = (totals[c] ?? 0) + 1;
  (filesPer[c] ??= new Set()).add(o.file);
}
const starred = out.filter((o) => o.cls.endsWith("*")).length;
let md = "| file | sites (line) | class | target |\n|---|---|---|---|\n";
const byFile = {};
for (const o of out) (byFile[o.file] ??= []).push(o);
for (const [f, os] of Object.entries(byFile)) {
  const groups = {};
  for (const o of os) (groups[`${o.cls}\u0000${o.note}`] ??= []).push(o.line);
  for (const [k, lines] of Object.entries(groups)) {
    const [cls, note] = k.split("\u0000");
    md += `| ${f} | ${lines.length} (L${lines.join(", L")}) | ${cls} | ${note} |\n`;
  }
}
const sum = Object.entries(totals).map(([c, n]) => `${c}: ${n} sites in ${filesPer[c].size} files`).join("; ");
const summary = `total ${out.length} sites in ${Object.keys(byFile).length} files. ${sum}. Of class b, ${starred} need a resolver not in phase 1.`;
fs.writeFileSync(`${outDir}/callsite-table.md`, md + "\n" + summary + "\n");
console.log(summary);
