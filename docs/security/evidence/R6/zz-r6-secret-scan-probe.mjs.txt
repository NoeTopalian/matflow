// R6 SF-82 probe against scripts/readiness/secret-scan.mjs scanText. Every secret-shaped value is generated at run
// time (nothing secret-shaped is written to this file); output carries rule, classification and the case label only.
import { randomBytes } from "node:crypto";
import { scanText } from "./scripts/readiness/secret-scan.mjs";

const rnd = (n) => randomBytes(n).toString("base64url").replace(/[-_]/g, "x").slice(0, n);
const ex = "exam" + "ple";
const cases = [
  // must be caught
  ["lower-case operator_password (env line)", `operator_password=${rnd(20)}`, "catch"],
  ["lower-case smtp_password (yaml)", `smtp_password: "${rnd(18)}"`, "catch"],
  ["mixed-case Smtp_Password (env)", `Smtp_Password=${rnd(18)}`, "catch"],
  ["SMTP_PASSWORD (env)", `SMTP_PASSWORD=${rnd(22)}`, "catch"],
  ["SMTP_PASSWORD quoted in a .ts source", `const cfg = { SMTP_PASSWORD: "${rnd(22)}" };`, "catch", "src/mail.ts"],
  ["password containing the marker word with a random stretch (env)", `SMTP_PASSWORD=${ex}${rnd(16)}`, "catch"],
  ["Neon owner URL whose password contains the marker word + random stretch", `DATABASE_URL=postgresql://neondb_owner:${ex}${rnd(14)}@ep-quiet-lake-a1b2c3d4.eu-west-2.aws.neon.tech/neondb`, "catch"],
  ["random stretch with the marker in the middle (token)", `api_token=${rnd(10)}${ex}${rnd(10)}`, "catch"],
  ["export prefix, lower-case name", `export db_password=${rnd(20)}`, "catch"],
  // must stay negative (placeholders / references)
  ["placeholder changeme", "SMTP_PASSWORD=changeme", "ignore"],
  ["bare marker word", `SMTP_PASSWORD=${ex}`, "ignore"],
  ["shell interpolation", "operator_password=${OPERATOR_PASSWORD}", "ignore"],
  ["windows env reference", "SMTP_PASSWORD=%SMTP_PASSWORD%", "ignore"],
  ["property reference in code", "const p = { SMTP_PASSWORD: process.env.SMTP_PASSWORD };", "ignore", "src/mail.ts"],
  ["localhost database URL", "DATABASE_URL=postgresql://postgres:postgres@localhost:5432/matflow", "ignore"],
  ["empty value", "SMTP_PASSWORD=", "ignore"],
  ["low-entropy cron secret", "CRON_SECRET=aaaaaaaaaaaaaaaa", "ignore"],
  ["zero-padded test key", "sk_test_" + "0".repeat(24), "ignore"],
];
const out = [];
for (const [label, text, want, file] of cases) {
  const f = scanText(text + "\n", file ?? ".env.local");
  const cand = f.filter((x) => x.classification === "candidate");
  const caught = cand.length > 0;
  out.push({ case: label, want, got: caught ? "candidate" : f.length ? f.map((x) => x.classification).join(",") : "none", rules: [...new Set(f.map((x) => x.rule))], ok: (want === "catch") === caught });
}
for (const o of out) console.log(JSON.stringify(o));
console.log(JSON.stringify({ cases: out.length, asExpected: out.filter((o) => o.ok).length }));
