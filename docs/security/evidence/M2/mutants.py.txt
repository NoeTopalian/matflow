# test command, record exit code and failing test names, restore the exact original bytes, verify by sha256.

OUT = sys.argv[1]
only = sys.argv[2:] if len(sys.argv) > 2 else None

def sha(b): return hashlib.sha256(b).hexdigest()

results = []
for m in M:
    if only and not any(m["id"].startswith(o) for o in only):
        continue
    path = os.path.join(REPO, m["file"])
    rec = {"id": m["id"], "file": m["file"]}
    if m.get("create"):
        assert not os.path.exists(path), "planted file already exists"
        open(path, "w", encoding="utf8", newline="").write(m["create"])
        orig = None
    else:
        orig = open(path, "rb").read()
        text = orig.decode("utf8")
        find, repl = m["find"], m["replace"]
        if text.count(find) == 0 and "\r\n" in text:  # the working tree is checked out with CRLF
            find, repl = find.replace("\n", "\r\n"), repl.replace("\n", "\r\n")
        n = text.count(find)
        if n != 1:
            rec["error"] = f"find occurs {n} times"; results.append(rec); continue
        open(path, "wb").write(text.replace(find, repl).encode("utf8"))
        rec["orig_sha16"] = sha(orig)[:16]
    try:
        p = subprocess.run(["bash", "-lc", f'source "<scratchpad>/security/M2/env.sh"; npx vitest run {m["test"]} 2>&1'],
                           capture_output=True, text=True, encoding="utf8", errors="replace", timeout=400)
        out = p.stdout
        rec["exit"] = p.returncode
        rec["failed"] = [l.strip() for l in out.splitlines() if l.strip().startswith("×")]
        rec["summary"] = [l.strip() for l in out.splitlines() if l.strip().startswith("Tests ")]
    finally:
        if orig is None:
            os.remove(path); rec["restored"] = not os.path.exists(path)
        else:
            open(path, "wb").write(orig)
            rec["restored"] = sha(open(path, "rb").read()) == sha(orig)
    results.append(rec)
    print(json.dumps({k: rec[k] for k in ("id", "exit", "restored")}), flush=True)

with open(OUT, "w", encoding="utf8") as f:
    for r in results: f.write(json.dumps(r, ensure_ascii=False) + "\n")
